// Function to Pay Bounty
async function payBounty(issueNumber, currency = 'USDC') {
  if (!config.bounties[issueNumber]) {
    console.error(`Bounty not defined for issue #${issueNumber}`);
    return;
  }
  
  const amount = config.bounties[issueNumber];
  const toAddress = 'RECIPIENT_WALLET_ADDRESS_HERE'; // Update with the recipient's wallet address
  
  if (currency === 'USDC') {

    const usdcContract = new ethers.Contract(usdcContractAddress, [
      'function transfer(address to, uint256 value) public returns (bool)',
    ], wallet);
    
    try {
      const tx = await usdcContract.transfer(toAddress, ethers.utils.parseUnits(amount.toString(), 6));
      console.log(`USDC Bounty paid for #${issueNumber}: TX Hash - ${tx.hash}`);

      console.error(`Error paying USDC Bounty for #${issueNumber}:`, error);
    }

    try {
      const tx = await wallet.sendTransaction({
        to: toAddress,
        value: ethers.utils.parseEther(amount.toString()),
      });
      console.log(`ETH Bounty paid for #${issueNumber}: TX Hash - ${tx.hash}`);
    } catch (error) {
      console.error(`Error paying ETH Bounty for #${issueNumber}:`, error);
    }
  } else {
    console.error(`Unsupported currency for bounty payment: ${currency}`);
  }
}
